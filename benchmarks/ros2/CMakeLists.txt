cmake_minimum_required(VERSION 3.16)
project(pipes_execution_pilot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
if(NOT EXISTS "${PIPES_LIBRARY}")
  message(FATAL_ERROR "Set PIPES_LIBRARY to the compiled libpipes_rs.so")
endif()
add_library(pipes_kernel SHARED IMPORTED)
set_target_properties(pipes_kernel PROPERTIES IMPORTED_LOCATION "${PIPES_LIBRARY}")
add_executable(execution-pilot composed.cpp)
# RoboStack's tracetools target exports bare -llttng names. Give the
# linker the package directory as well as CMake's package search prefix.
find_library(LTTNG_UST_LIBRARY NAMES lttng-ust HINTS "$ENV{CONDA_PREFIX}/lib" REQUIRED)
get_filename_component(LTTNG_LIBRARY_DIR "${LTTNG_UST_LIBRARY}" DIRECTORY)
target_link_directories(execution-pilot PRIVATE "${LTTNG_LIBRARY_DIR}")
ament_target_dependencies(execution-pilot rclcpp std_msgs)
target_link_libraries(execution-pilot pipes_kernel)
target_compile_options(execution-pilot PRIVATE -Wall -Wextra)
